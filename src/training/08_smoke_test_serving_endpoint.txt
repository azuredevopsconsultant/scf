# Databricks notebook source
# MAGIC %md
# MAGIC # 08 - Smoke Test Serving Endpoint
# MAGIC Sends one record to the serving endpoint and asserts a non-empty response.

# COMMAND ----------
dbutils.widgets.text("catalog", "poc_mlops_dev")
dbutils.widgets.text("schema", "savings_cashflow")
dbutils.widgets.text("serving_endpoint_name", "scf-cohort-serving-endpoint-dev")

catalog = dbutils.widgets.get("catalog")
schema = dbutils.widgets.get("schema")
serving_endpoint_name = dbutils.widgets.get("serving_endpoint_name")

from databricks.sdk import WorkspaceClient
import pandas as pd
import sys

sys.path.append("../..")
from src.common.config import get_config

cfg = get_config(catalog, schema)

sample_pdf = (
    spark.table(cfg.silver_agg_cohort)
    .select(
        "product",
        "cohort",
        "reporting_period",
        "months_since_start",
        "dbb_range",
        "balance_lag_1",
    )
    .limit(1)
    .toPandas()
)

assert not sample_pdf.empty, "No rows available for serving smoke test"

sample_pdf["cohort"] = sample_pdf["cohort"].astype(str)
sample_pdf["reporting_period"] = sample_pdf["reporting_period"].astype(str)
sample_pdf["months_since_start"] = pd.to_numeric(sample_pdf["months_since_start"], errors="coerce").fillna(0).astype(int)
sample_pdf["balance_lag_1"] = pd.to_numeric(sample_pdf["balance_lag_1"], errors="coerce").fillna(0.0).astype(float)
sample_pdf["dbb_range"] = sample_pdf["dbb_range"].astype(str)

records = sample_pdf.to_dict(orient="records")

w = WorkspaceClient()
response = w.serving_endpoints.query(name=serving_endpoint_name, dataframe_records=records)

predictions = getattr(response, "predictions", None)
if predictions is None:
    predictions = getattr(response, "data", None)

assert predictions is not None, "Serving endpoint returned no predictions payload"
try:
    prediction_count = len(predictions)
except Exception:
    prediction_count = 1

assert prediction_count > 0, "Serving endpoint returned empty predictions"
print(f"Serving smoke test passed for {serving_endpoint_name}; rows={prediction_count}")

dbutils.jobs.taskValues.set(key="smoke_test_passed", value=True)
dbutils.jobs.taskValues.set(key="smoke_test_prediction_count", value=int(prediction_count))
