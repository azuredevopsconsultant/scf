# MAGIC # 06 - Confirmation Mail
# MAGIC Mirrors `src/inference/07_confirmation_mail.py`: always runs
# MAGIC (`run_if: ALL_DONE`), summarises the run (products evaluated, portfolio
# MAGIC MAPE, and whether the challenger was promoted to champion) in one email.

# COMMAND ----------
dbutils.widgets.text("catalog", "poc_mlops_dev")
dbutils.widgets.text("schema", "savings_cashflow")
dbutils.widgets.text("notification_email", "")
dbutils.widgets.text("notification_webhook_url", "")
catalog = dbutils.widgets.get("catalog")
schema = dbutils.widgets.get("schema")
notification_email = dbutils.widgets.get("notification_email")
notification_webhook_url = dbutils.widgets.get("notification_webhook_url")

import sys
sys.path.append("../..")
from src.common.notifications import send_confirmation_mail, send_webhook_notification

# COMMAND ----------
def safe_task_value(task_key, key, default="N/A"):
    try:
        return dbutils.jobs.taskValues.get(taskKey=task_key, key=key, default=default)
    except Exception:
        return default

summary = {
    "Products evaluated": safe_task_value("model_evaluation", "n_products_evaluated"),
    "Portfolio MAPE (balance)": safe_task_value("model_evaluation", "portfolio_mape"),
    "Registered version": safe_task_value("model_registration", "registered_version"),
    "Promoted to champion": safe_task_value("model_registration", "promoted"),
    "Promotion reason": safe_task_value("model_registration", "promotion_reason"),
    "Serving endpoint": safe_task_value("deploy_serving_endpoint", "serving_endpoint_name"),
    "Serving model version": safe_task_value("deploy_serving_endpoint", "serving_model_version"),
    "Serving alias used": safe_task_value("deploy_serving_endpoint", "serving_alias_used"),
    "Serving action": safe_task_value("deploy_serving_endpoint", "serving_action"),
    "Serving smoke test passed": safe_task_value("smoke_test_serving_endpoint", "smoke_test_passed"),
    "Serving smoke rows": safe_task_value("smoke_test_serving_endpoint", "smoke_test_prediction_count"),
}

run_url = dbutils.notebook.entry_point.getDbutils().notebook().getContext().apiUrl().get()

# COMMAND ----------
if notification_email:
    send_confirmation_mail(
        to_addresses=[notification_email],
        environment=catalog,
        pipeline_name="training",
        summary=summary,
        run_url=run_url,
    )
    print(f"Confirmation mail sent to {notification_email}")
else:
    print("No notification_email configured - skipping confirmation mail.")

if notification_webhook_url:
    send_webhook_notification(
        webhook_url=notification_webhook_url,
        title=f"[{catalog}] training pipeline summary",
        facts={k: str(v) for k, v in summary.items()},
        run_url=run_url,
    )
    print("Confirmation webhook sent")

print(summary)
