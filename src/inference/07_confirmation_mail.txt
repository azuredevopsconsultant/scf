# Databricks notebook source
# MAGIC %md
# MAGIC # 07 - Confirmation Mail
# MAGIC Always runs (`run_if: ALL_DONE` in the job spec) so the recipient gets
# MAGIC a pass/fail summary of every upstream stage in one email, regardless of
# MAGIC where in the DAG something broke - this is deliberately separate from
# MAGIC the job-level on_success/on_failure notification, which only tells you
# MAGIC the run outcome, not *what happened at each stage*.

# COMMAND ----------
dbutils.widgets.text("catalog", "poc_mlops_dev")
dbutils.widgets.text("schema", "savings_cashflow")
dbutils.widgets.text("notification_email", "")
dbutils.widgets.text("notification_webhook_url", "")
dbutils.widgets.text("pipeline_name", "inference")
catalog = dbutils.widgets.get("catalog")
schema = dbutils.widgets.get("schema")
notification_email = dbutils.widgets.get("notification_email")
notification_webhook_url = dbutils.widgets.get("notification_webhook_url")
pipeline_name = dbutils.widgets.get("pipeline_name")

import sys
sys.path.append("../..")
from src.common.notifications import send_confirmation_mail, send_webhook_notification

# COMMAND ----------
def safe_task_value(task_key, key, default="N/A"):
    try:
        return dbutils.jobs.taskValues.get(taskKey=task_key, key=key, default=default)
    except Exception:
        return default

summary = {
    "Predictions written": safe_task_value("inferencing", "n_predictions"),
    "Drift detected": safe_task_value("data_drift", "drift_detected"),
    "Data quality passed": safe_task_value("data_quality", "data_quality_passed"),
    "Retraining recommended": safe_task_value("retraining_trigger", "retrain_recommended"),
    "Retraining reason": safe_task_value("retraining_trigger", "retrain_reason"),
}

run_url = dbutils.notebook.entry_point.getDbutils().notebook().getContext().apiUrl().get()

# COMMAND ----------
if notification_email:
    send_confirmation_mail(
        to_addresses=[notification_email],
        environment=catalog,
        pipeline_name=pipeline_name,
        summary=summary,
        run_url=run_url,
    )
    print(f"Confirmation mail sent to {notification_email}")
else:
    print("No notification_email configured - skipping confirmation mail.")

if notification_webhook_url:
    send_webhook_notification(
        webhook_url=notification_webhook_url,
        title=f"[{catalog}] {pipeline_name} pipeline summary",
        facts={k: str(v) for k, v in summary.items()},
        run_url=run_url,
    )
    print("Confirmation webhook sent")

print(summary)
